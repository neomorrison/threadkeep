Write a Python decorator that retries a flaky function with exponential backoff.
Here is a small decorator that retries with backoff and jitter:
def retry(times=5, base=0.5):
    def wrap(fn):
        def inner(*a, **kw):
            for i in range(times):
                try:
                    return fn(*a, **kw)
                except Exception:
                    if i == times - 1:
                        raise
                    time.sleep(base * 2 ** i)
        return inner
    return wrap
